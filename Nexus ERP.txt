import React, { useState, useEffect } from 'react';
import { 
  LayoutDashboard, 
  ShoppingCart, 
  Package, 
  FileText, 
  Landmark, 
  Settings, 
  Bell, 
  Search, 
  Plus, 
  CheckCircle2, 
  AlertTriangle, 
  Server, 
  QrCode, 
  ChevronRight, 
  Download, 
  RefreshCw,
  X
} from 'lucide-react';

// --- Mock Data & Constants ---

const INITIAL_INVENTORY = [
  { id: 'ITM-001', name: 'Premium Widget A', price: 1500, taxRate: 0.16, stock: 45, category: 'Hardware' },
  { id: 'ITM-002', name: 'Service Plan B', price: 5000, taxRate: 0.16, stock: 999, category: 'Services' },
  { id: 'ITM-003', name: 'Zero-Rated Grain', price: 200, taxRate: 0.00, stock: 1500, category: 'Foodstuff' },
];

const INITIAL_CUSTOMERS = [
  { id: 'CUST-01', name: 'Acme Logistics Ltd', pin: 'P051123456Z' },
  { id: 'CUST-02', name: 'Global Traders Inc', pin: 'A002345678Y' },
];

const MOCK_COMPANY_INFO = {
  name: "TechNova Systems Ltd",
  pin: "P001234567X",
  address: "123 Moi Avenue, Nairobi",
  etimsDeviceID: "KRA-VSCU-8892"
};

// --- Helper Components ---

const Card = ({ children, className = "" }) => (
  <div className={`bg-white rounded-xl border border-slate-200 shadow-sm ${className}`}>
    {children}
  </div>
);

const Badge = ({ children, variant = "neutral" }) => {
  const styles = {
    neutral: "bg-slate-100 text-slate-700 border-slate-200",
    success: "bg-emerald-50 text-emerald-700 border-emerald-200",
    warning: "bg-amber-50 text-amber-700 border-amber-200",
    danger: "bg-rose-50 text-rose-700 border-rose-200",
    kra: "bg-indigo-50 text-indigo-700 border-indigo-200",
  };
  return (
    <span className={`px-2.5 py-1 rounded-full text-xs font-medium border ${styles[variant]}`}>
      {children}
    </span>
  );
};

// --- Main Application ---

export default function App() {
  const [activeView, setActiveView] = useState('dashboard');
  const [inventory, setInventory] = useState(INITIAL_INVENTORY);
  const [invoices, setInvoices] = useState([]);
  const [journalEntries, setJournalEntries] = useState([]);
  const [notifications, setNotifications] = useState([]);
  
  // Create Invoice State
  const [cart, setCart] = useState([]);
  const [selectedCustomer, setSelectedCustomer] = useState(INITIAL_CUSTOMERS[0].id);

  // --- Core ERP Logic ---

  // 1. Accounting: Post to General Ledger
  const postToLedger = (invoice) => {
    const total = invoice.total;
    const tax = invoice.taxAmount;
    const revenue = total - tax;
    
    const newEntry = {
      id: `JRN-${Date.now()}`,
      date: new Date().toISOString().split('T')[0],
      reference: `INV-${invoice.id}`,
      description: `Sales Invoice #${invoice.id} - ${invoice.customerName}`,
      lines: [
        { account: 'Accounts Receivable', debit: total, credit: 0 },
        { account: 'Sales Revenue', debit: 0, credit: revenue },
        { account: 'VAT Liability (Output)', debit: 0, credit: tax },
      ]
    };
    setJournalEntries(prev => [newEntry, ...prev]);
  };

  // 2. eTIMS: Simulate Transmission
  const transmitToKRA = (invoiceId) => {
    // Find invoice
    const invoice = invoices.find(inv => inv.id === invoiceId);
    if (!invoice) return;

    // Simulate API delay
    setTimeout(() => {
      const updatedInvoices = invoices.map(inv => {
        if (inv.id === invoiceId) {
          return {
            ...inv,
            etimsStatus: 'Verified',
            etimsData: {
              controlUnitId: MOCK_COMPANY_INFO.etimsDeviceID,
              receiptLabel: `CU-${Math.floor(Math.random() * 10000)}`,
              date: new Date().toLocaleString(),
              signature: Array(16).fill(0).map(() => Math.random().toString(36)[2]).join('').toUpperCase()
            }
          };
        }
        return inv;
      });
      setInvoices(updatedInvoices);
      
      // Add notification
      setNotifications(prev => [{
        id: Date.now(), 
        title: 'eTIMS Success', 
        msg: `Invoice #${invoiceId} successfully fiscalized.`
      }, ...prev]);

    }, 1500);
  };

  const handleCreateInvoice = () => {
    const customer = INITIAL_CUSTOMERS.find(c => c.id === selectedCustomer);
    const subtotal = cart.reduce((acc, item) => acc + (item.price * item.qty), 0);
    const taxAmount = cart.reduce((acc, item) => acc + (item.price * item.qty * item.taxRate), 0);
    const total = subtotal + taxAmount;

    const newInvoice = {
      id: Date.now().toString().slice(-6), // Simple ID
      date: new Date().toISOString().split('T')[0],
      customerId: customer.id,
      customerName: customer.name,
      customerPin: customer.pin,
      items: [...cart],
      subtotal,
      taxAmount,
      total,
      status: 'Draft',
      etimsStatus: 'Pending',
      etimsData: null
    };

    setInvoices([newInvoice, ...invoices]);
    setCart([]);
    postToLedger(newInvoice); // Auto-post to accounting
    setActiveView('etims'); // Redirect to eTIMS portal to show flow
  };

  // --- Views ---

  const Sidebar = () => (
    <div className="w-64 bg-slate-900 text-slate-300 flex flex-col fixed h-full z-20">
      <div className="p-6">
        <div className="flex items-center gap-3 text-white mb-8">
          <div className="bg-emerald-500 p-2 rounded-lg">
            <Server size={20} />
          </div>
          <div>
            <h1 className="font-bold text-lg leading-tight">Nexus ERP</h1>
            <p className="text-xs text-slate-500">v2.1 • KRA Ready</p>
          </div>
        </div>

        <nav className="space-y-1">
          {[
            { id: 'dashboard', icon: LayoutDashboard, label: 'Dashboard' },
            { id: 'sales', icon: ShoppingCart, label: 'Sales & Invoicing' },
            { id: 'inventory', icon: Package, label: 'Inventory' },
            { id: 'accounting', icon: Landmark, label: 'Accounting' },
            { id: 'etims', icon: QrCode, label: 'eTIMS Portal' },
            { id: 'settings', icon: Settings, label: 'Settings' },
          ].map((item) => (
            <button
              key={item.id}
              onClick={() => setActiveView(item.id)}
              className={`w-full flex items-center gap-3 px-4 py-3 rounded-lg transition-colors ${
                activeView === item.id 
                  ? 'bg-emerald-600 text-white' 
                  : 'hover:bg-slate-800 hover:text-white'
              }`}
            >
              <item.icon size={18} />
              <span className="font-medium text-sm">{item.label}</span>
            </button>
          ))}
        </nav>
      </div>
      
      <div className="mt-auto p-6 border-t border-slate-800">
        <div className="bg-slate-800 rounded-lg p-3">
          <p className="text-xs text-slate-400 uppercase font-semibold mb-2">System Status</p>
          <div className="flex items-center gap-2 mb-1">
            <div className="w-2 h-2 rounded-full bg-emerald-500 animate-pulse"></div>
            <span className="text-xs text-white">Database: Online</span>
          </div>
          <div className="flex items-center gap-2">
            <div className="w-2 h-2 rounded-full bg-emerald-500"></div>
            <span className="text-xs text-white">KRA OSCU: Connected</span>
          </div>
        </div>
      </div>
    </div>
  );

  const Dashboard = () => {
    const totalSales = invoices.reduce((acc, curr) => acc + curr.total, 0);
    const pendingEtims = invoices.filter(i => i.etimsStatus === 'Pending').length;

    return (
      <div className="space-y-6">
        <h2 className="text-2xl font-bold text-slate-800">Executive Overview</h2>
        
        {/* Stats Grid */}
        <div className="grid grid-cols-1 md:grid-cols-4 gap-4">
          <Card className="p-5">
            <p className="text-slate-500 text-sm font-medium">Total Revenue</p>
            <h3 className="text-2xl font-bold text-slate-900 mt-1">KES {totalSales.toLocaleString()}</h3>
            <span className="text-emerald-600 text-xs font-medium flex items-center gap-1 mt-2">
              <RefreshCw size={12} /> Real-time
            </span>
          </Card>
          <Card className="p-5">
            <p className="text-slate-500 text-sm font-medium">Pending Fiscalization</p>
            <h3 className="text-2xl font-bold text-slate-900 mt-1">{pendingEtims}</h3>
            {pendingEtims > 0 ? (
              <span className="text-amber-600 text-xs font-medium flex items-center gap-1 mt-2">
                <AlertTriangle size={12} /> Action Required
              </span>
            ) : (
              <span className="text-emerald-600 text-xs font-medium flex items-center gap-1 mt-2">
                <CheckCircle2 size={12} /> All Compliant
              </span>
            )}
          </Card>
          <Card className="p-5">
            <p className="text-slate-500 text-sm font-medium">Low Stock Alerts</p>
            <h3 className="text-2xl font-bold text-slate-900 mt-1">2</h3>
            <span className="text-slate-400 text-xs mt-2 block">Items below threshold</span>
          </Card>
          <Card className="p-5">
            <p className="text-slate-500 text-sm font-medium">VAT Liability</p>
            <h3 className="text-2xl font-bold text-slate-900 mt-1">
              KES {invoices.reduce((acc, i) => acc + i.taxAmount, 0).toLocaleString()}
            </h3>
            <span className="text-slate-400 text-xs mt-2 block">Current Month</span>
          </Card>
        </div>

        {/* Recent Activity */}
        <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
          <Card className="p-6">
            <h3 className="font-semibold text-slate-800 mb-4">Recent Transactions</h3>
            <div className="space-y-4">
              {invoices.slice(0, 5).map(inv => (
                <div key={inv.id} className="flex justify-between items-center border-b border-slate-100 pb-3 last:border-0">
                  <div>
                    <p className="font-medium text-slate-900">Inv #{inv.id}</p>
                    <p className="text-xs text-slate-500">{inv.customerName}</p>
                  </div>
                  <div className="text-right">
                    <p className="font-bold text-slate-900">KES {inv.total.toLocaleString()}</p>
                    <p className="text-xs text-slate-500">{inv.date}</p>
                  </div>
                </div>
              ))}
              {invoices.length === 0 && <p className="text-slate-400 text-sm italic">No transactions yet.</p>}
            </div>
          </Card>

          <Card className="p-6 bg-slate-900 text-white border-none">
            <h3 className="font-semibold mb-4 flex items-center gap-2">
              <Server size={18} className="text-emerald-400"/>
              eTIMS Status Monitor
            </h3>
            <div className="space-y-4">
              <div className="flex justify-between text-sm">
                <span className="text-slate-400">Device Status</span>
                <span className="text-emerald-400 font-mono">ONLINE</span>
              </div>
              <div className="flex justify-between text-sm">
                <span className="text-slate-400">Last Sync</span>
                <span className="text-slate-200 font-mono">{new Date().toLocaleTimeString()}</span>
              </div>
              <div className="flex justify-between text-sm">
                <span className="text-slate-400">Environment</span>
                <span className="text-amber-400 font-mono">SANDBOX</span>
              </div>
              <div className="mt-8 pt-6 border-t border-slate-700">
                <p className="text-xs text-slate-400 mb-2">Compliance Queue</p>
                <div className="w-full bg-slate-800 h-2 rounded-full overflow-hidden">
                  <div 
                    className="bg-emerald-500 h-full transition-all duration-500" 
                    style={{ width: `${invoices.length > 0 ? (invoices.filter(i => i.etimsStatus === 'Verified').length / invoices.length) * 100 : 100}%` }}
                  ></div>
                </div>
              </div>
            </div>
          </Card>
        </div>
      </div>
    );
  };

  const SalesInvoicing = () => {
    const addToCart = (itemId) => {
      const item = inventory.find(i => i.id === itemId);
      const existing = cart.find(c => c.itemId === itemId);
      
      if (existing) {
        setCart(cart.map(c => c.itemId === itemId ? { ...c, qty: c.qty + 1 } : c));
      } else {
        setCart([...cart, { itemId: item.id, name: item.name, price: item.price, taxRate: item.taxRate, qty: 1 }]);
      }
    };

    return (
      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6 h-[calc(100vh-140px)]">
        {/* Product Catalog */}
        <Card className="lg:col-span-2 flex flex-col overflow-hidden">
          <div className="p-4 border-b border-slate-200 bg-slate-50 flex justify-between items-center">
            <h3 className="font-semibold text-slate-800">Product Catalog</h3>
            <div className="relative w-64">
              <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-slate-400" size={16} />
              <input type="text" placeholder="Search items..." className="w-full pl-9 pr-3 py-1.5 text-sm border border-slate-300 rounded-md outline-none focus:ring-2 focus:ring-emerald-500" />
            </div>
          </div>
          <div className="overflow-y-auto p-4 grid grid-cols-2 md:grid-cols-3 gap-4">
            {inventory.map(item => (
              <button 
                key={item.id} 
                onClick={() => addToCart(item.id)}
                className="flex flex-col items-start p-4 rounded-xl border border-slate-200 hover:border-emerald-500 hover:shadow-md transition-all text-left bg-white group"
              >
                <div className="w-full flex justify-between items-start mb-2">
                  <span className="text-xs font-bold text-slate-400 uppercase tracking-wider">{item.category}</span>
                  <span className="text-emerald-600 bg-emerald-50 px-2 py-0.5 rounded text-xs font-medium">Stock: {item.stock}</span>
                </div>
                <h4 className="font-semibold text-slate-800 mb-1 group-hover:text-emerald-700">{item.name}</h4>
                <p className="text-slate-500 text-sm">KES {item.price.toLocaleString()}</p>
                <span className="text-xs text-slate-400 mt-2">Tax: {(item.taxRate * 100).toFixed(0)}%</span>
              </button>
            ))}
          </div>
        </Card>

        {/* Invoice Builder */}
        <Card className="flex flex-col h-full bg-slate-50">
          <div className="p-4 border-b border-slate-200 bg-white">
            <h3 className="font-semibold text-slate-800 flex items-center gap-2">
              <FileText size={18} className="text-emerald-600"/> 
              New Invoice
            </h3>
          </div>
          
          <div className="p-4 bg-white border-b border-slate-200">
            <label className="block text-xs font-semibold text-slate-500 mb-1">Customer</label>
            <select 
              className="w-full p-2 border border-slate-300 rounded-md text-sm outline-none focus:border-emerald-500"
              value={selectedCustomer}
              onChange={(e) => setSelectedCustomer(e.target.value)}
            >
              {INITIAL_CUSTOMERS.map(c => (
                <option key={c.id} value={c.id}>{c.name} ({c.pin})</option>
              ))}
            </select>
          </div>

          <div className="flex-1 overflow-y-auto p-4 space-y-3">
            {cart.length === 0 ? (
              <div className="h-full flex flex-col items-center justify-center text-slate-400 text-sm text-center">
                <ShoppingCart size={40} className="mb-2 opacity-20"/>
                <p>Cart is empty.<br/>Select items to begin.</p>
              </div>
            ) : (
              cart.map((item, idx) => (
                <div key={idx} className="flex justify-between items-center bg-white p-3 rounded-lg border border-slate-200 shadow-sm">
                  <div>
                    <p className="font-medium text-sm text-slate-800">{item.name}</p>
                    <p className="text-xs text-slate-500">{item.qty} x {item.price.toLocaleString()}</p>
                  </div>
                  <div className="text-right">
                    <p className="font-medium text-sm text-slate-800">{(item.qty * item.price).toLocaleString()}</p>
                    <button 
                      onClick={() => setCart(cart.filter((_, i) => i !== idx))}
                      className="text-xs text-rose-500 hover:underline mt-1"
                    >
                      Remove
                    </button>
                  </div>
                </div>
              ))
            )}
          </div>

          <div className="p-6 bg-white border-t border-slate-200 shadow-up">
            <div className="space-y-2 mb-4 text-sm">
              <div className="flex justify-between text-slate-600">
                <span>Subtotal</span>
                <span>{cart.reduce((acc, i) => acc + (i.price * i.qty), 0).toLocaleString()}</span>
              </div>
              <div className="flex justify-between text-slate-600">
                <span>VAT (16%)</span>
                <span>{cart.reduce((acc, i) => acc + (i.price * i.qty * i.taxRate), 0).toLocaleString()}</span>
              </div>
              <div className="flex justify-between font-bold text-lg text-slate-900 border-t border-slate-100 pt-2">
                <span>Total</span>
                <span>KES {cart.reduce((acc, i) => acc + (i.price * i.qty * (1 + i.taxRate)), 0).toLocaleString()}</span>
              </div>
            </div>
            <button 
              onClick={handleCreateInvoice}
              disabled={cart.length === 0}
              className="w-full py-3 bg-emerald-600 hover:bg-emerald-700 disabled:bg-slate-300 disabled:cursor-not-allowed text-white font-medium rounded-lg transition-colors flex items-center justify-center gap-2"
            >
              <CheckCircle2 size={18} />
              Process Invoice
            </button>
          </div>
        </Card>
      </div>
    );
  };

  const AccountingView = () => (
    <div className="space-y-6">
      <div className="flex justify-between items-end">
        <div>
          <h2 className="text-2xl font-bold text-slate-800">General Ledger</h2>
          <p className="text-slate-500">Automated double-entry logs from operations</p>
        </div>
        <button className="px-4 py-2 border border-slate-300 bg-white rounded-lg text-sm font-medium text-slate-600 hover:bg-slate-50 flex items-center gap-2">
          <Download size={16} /> Export CSV
        </button>
      </div>

      <Card className="overflow-hidden">
        <table className="w-full text-left text-sm text-slate-600">
          <thead className="bg-slate-50 text-slate-900 font-semibold border-b border-slate-200">
            <tr>
              <th className="px-6 py-3">Date</th>
              <th className="px-6 py-3">Journal ID</th>
              <th className="px-6 py-3">Account</th>
              <th className="px-6 py-3">Description</th>
              <th className="px-6 py-3 text-right">Debit</th>
              <th className="px-6 py-3 text-right">Credit</th>
            </tr>
          </thead>
          <tbody className="divide-y divide-slate-100">
            {journalEntries.length > 0 ? (
              journalEntries.map((entry) => (
                <React.Fragment key={entry.id}>
                  {entry.lines.map((line, idx) => (
                    <tr key={`${entry.id}-${idx}`} className={idx === 0 ? "bg-white" : "bg-slate-50/50"}>
                      <td className="px-6 py-2">{idx === 0 ? entry.date : ''}</td>
                      <td className="px-6 py-2 font-mono text-xs">{idx === 0 ? entry.id : ''}</td>
                      <td className="px-6 py-2 font-medium text-slate-700">{line.account}</td>
                      <td className="px-6 py-2 text-slate-500">{idx === 0 ? entry.description : ''}</td>
                      <td className="px-6 py-2 text-right font-mono">{line.debit > 0 ? line.debit.toLocaleString() : '-'}</td>
                      <td className="px-6 py-2 text-right font-mono">{line.credit > 0 ? line.credit.toLocaleString() : '-'}</td>
                    </tr>
                  ))}
                  <tr className="bg-slate-100 h-px"><td colSpan={6}></td></tr>
                </React.Fragment>
              ))
            ) : (
              <tr>
                <td colSpan="6" className="px-6 py-12 text-center text-slate-400">
                  No accounting entries generated yet. Process a sale to see ledger activity.
                </td>
              </tr>
            )}
          </tbody>
        </table>
      </Card>
    </div>
  );

  const ETimsPortal = () => (
    <div className="space-y-6">
      <div className="bg-indigo-600 rounded-xl p-8 text-white relative overflow-hidden shadow-lg">
        <div className="relative z-10 max-w-2xl">
          <h2 className="text-3xl font-bold mb-2">eTIMS Compliance Hub</h2>
          <p className="text-indigo-100 mb-6">
            Manage your fiscal obligations. Ensure all invoices are signed and transmitted to KRA 
            within 24 hours of generation to avoid penalties.
          </p>
          <div className="flex gap-4">
            <div className="bg-white/10 backdrop-blur-sm p-3 rounded-lg border border-white/20">
              <p className="text-xs uppercase tracking-wider opacity-70">Trader PIN</p>
              <p className="font-mono font-bold text-lg">{MOCK_COMPANY_INFO.pin}</p>
            </div>
            <div className="bg-white/10 backdrop-blur-sm p-3 rounded-lg border border-white/20">
              <p className="text-xs uppercase tracking-wider opacity-70">Device ID</p>
              <p className="font-mono font-bold text-lg">{MOCK_COMPANY_INFO.etimsDeviceID}</p>
            </div>
          </div>
        </div>
        {/* Decorative BG */}
        <QrCode className="absolute -right-10 -bottom-10 text-indigo-500 opacity-20" size={250} />
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        <div className="lg:col-span-2 space-y-4">
          <h3 className="font-bold text-slate-800 text-lg">Transmission Queue</h3>
          <div className="space-y-3">
            {invoices.length > 0 ? (
              invoices.map(inv => (
                <Card key={inv.id} className="p-4 flex flex-col sm:flex-row items-center justify-between gap-4">
                  <div className="flex items-center gap-4">
                    <div className={`p-3 rounded-full ${inv.etimsStatus === 'Verified' ? 'bg-emerald-100 text-emerald-600' : 'bg-amber-100 text-amber-600'}`}>
                      {inv.etimsStatus === 'Verified' ? <CheckCircle2 size={24}/> : <Server size={24}/>}
                    </div>
                    <div>
                      <div className="flex items-center gap-2">
                        <h4 className="font-bold text-slate-800">Invoice #{inv.id}</h4>
                        <Badge variant={inv.etimsStatus === 'Verified' ? 'success' : 'warning'}>
                          {inv.etimsStatus === 'Verified' ? 'Fiscalized' : 'Pending Transmission'}
                        </Badge>
                      </div>
                      <p className="text-sm text-slate-500">
                        {inv.date} • {inv.customerName} • <span className="font-mono">KES {inv.total.toLocaleString()}</span>
                      </p>
                      {inv.etimsStatus === 'Verified' && (
                        <p className="text-xs text-slate-400 mt-1 font-mono">
                          CU: {inv.etimsData.receiptLabel} | SIG: {inv.etimsData.signature.slice(0, 8)}...
                        </p>
                      )}
                    </div>
                  </div>
                  
                  {inv.etimsStatus === 'Pending' ? (
                    <button 
                      onClick={() => transmitToKRA(inv.id)}
                      className="px-4 py-2 bg-indigo-600 hover:bg-indigo-700 text-white rounded-lg text-sm font-medium transition-colors flex items-center gap-2 whitespace-nowrap"
                    >
                      <RefreshCw size={16} /> Transmit to KRA
                    </button>
                  ) : (
                    <button className="px-4 py-2 border border-slate-200 text-slate-400 rounded-lg text-sm font-medium cursor-default flex items-center gap-2">
                      <CheckCircle2 size={16} /> Synced
                    </button>
                  )}
                </Card>
              ))
            ) : (
              <div className="text-center py-12 bg-slate-50 rounded-xl border border-dashed border-slate-300">
                <p className="text-slate-400">No invoices to transmit.</p>
              </div>
            )}
          </div>
        </div>

        <div className="space-y-6">
          <Card className="p-5">
            <h3 className="font-semibold text-slate-800 mb-3">Debug Log (Simulation)</h3>
            <div className="bg-slate-900 rounded-lg p-4 font-mono text-xs text-green-400 h-64 overflow-y-auto">
              <p className="opacity-50">{'// KRA OSCU Connection Log'}</p>
              <p>{'> Initializing VSCU...'}</p>
              <p>{'> Connection established (wss://etims-api-sbx.kra.go.ke)'}</p>
              <p>{'> Ready for transmission.'}</p>
              {invoices.filter(i => i.etimsStatus === 'Verified').map(i => (
                 <div key={i.id} className="mt-2 border-t border-slate-700 pt-2">
                   <p className="text-white">{'> POST /transmitInvoice'}</p>
                   <p>Payload: &#123;</p>
                   <p className="pl-2">"invID": "{i.id}",</p>
                   <p className="pl-2">"custPin": "{i.customerPin}",</p>
                   <p className="pl-2">"total": {i.total},</p>
                   <p className="pl-2">"tax": {i.taxAmount}</p>
                   <p>&#125;</p>
                   <p className="text-yellow-300"><br/>&lt; RESPONSE 200 OK</p>
                   <p className="pl-2 text-yellow-300">"signature": "{i.etimsData.signature}"</p>
                 </div>
              ))}
            </div>
          </Card>
        </div>
      </div>
    </div>
  );

  return (
    <div className="min-h-screen bg-slate-50 font-sans flex">
      <Sidebar />
      
      {/* Main Content Area */}
      <div className="flex-1 ml-64">
        {/* Header */}
        <header className="h-16 bg-white border-b border-slate-200 flex items-center justify-between px-8 sticky top-0 z-10">
          <div className="flex items-center gap-4">
            <h2 className="text-xl font-bold text-slate-800 capitalize">
              {activeView === 'etims' ? 'eTIMS Portal' : activeView.replace('-', ' ')}
            </h2>
          </div>
          <div className="flex items-center gap-6">
            <div className="relative">
              <Bell size={20} className="text-slate-400 hover:text-slate-600 cursor-pointer transition-colors" />
              {notifications.length > 0 && (
                <span className="absolute -top-1 -right-1 w-2.5 h-2.5 bg-rose-500 rounded-full border border-white"></span>
              )}
            </div>
            <div className="flex items-center gap-3 pl-6 border-l border-slate-200">
              <div className="text-right hidden sm:block">
                <p className="text-sm font-medium text-slate-700">Admin User</p>
                <p className="text-xs text-slate-500">Finance Manager</p>
              </div>
              <div className="w-9 h-9 bg-emerald-100 text-emerald-700 rounded-full flex items-center justify-center font-bold">
                AU
              </div>
            </div>
          </div>
        </header>

        {/* View Container */}
        <main className="p-8">
          {activeView === 'dashboard' && <Dashboard />}
          {activeView === 'sales' && <SalesInvoicing />}
          {activeView === 'accounting' && <AccountingView />}
          {activeView === 'etims' && <ETimsPortal />}
          
          {/* Fallback for empty views */}
          {(activeView === 'inventory' || activeView === 'settings') && (
            <div className="flex flex-col items-center justify-center h-96 text-slate-400">
              <Package size={48} className="mb-4 opacity-20" />
              <p>Module under development for this demo.</p>
              <button onClick={() => setActiveView('dashboard')} className="mt-4 text-emerald-600 hover:underline">Return Home</button>
            </div>
          )}
        </main>
      </div>

      {/* Notifications Toast */}
      <div className="fixed bottom-6 right-6 space-y-2 z-50">
        {notifications.map((notif, i) => (
          <div key={notif.id} className="bg-slate-800 text-white p-4 rounded-lg shadow-xl flex items-start gap-3 w-80 animate-in slide-in-from-right duration-300">
            <CheckCircle2 className="text-emerald-400 shrink-0 mt-0.5" size={18} />
            <div className="flex-1">
              <h4 className="font-semibold text-sm">{notif.title}</h4>
              <p className="text-xs text-slate-300 mt-1">{notif.msg}</p>
            </div>
            <button 
              onClick={() => setNotifications(prev => prev.filter(n => n.id !== notif.id))}
              className="text-slate-500 hover:text-white"
            >
              <X size={14} />
            </button>
          </div>
        ))}
      </div>
    </div>
  );
}
